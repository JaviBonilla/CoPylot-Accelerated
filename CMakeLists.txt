cmake_minimum_required(VERSION 3.24)
project(copilot_headless LANGUAGES C CXX)
option(BUILD_COPILOT_API "Build the headless CoPylot shared library" ON)
option(SOLARPILOT_ENABLE_CORE_PROFILE "Instrument selected core scopes" OFF)
option(SOLARPILOT_ENABLE_FLUX_DIAGNOSTICS "Enable internal legacy/flat differential controls" OFF)
option(SOLARPILOT_ENABLE_CUDA "Enable hybrid CUDA Hermite flux density and device probe" OFF)
option(SOLARPILOT_ENABLE_SANITIZERS "Enable CPU Address/Undefined sanitizers" OFF)
if(SOLARPILOT_ENABLE_SANITIZERS)
  if(SOLARPILOT_ENABLE_CUDA OR MSVC)
    message(FATAL_ERROR "Sanitizer build is CPU-only and requires GCC/Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(SOLARPILOT_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "SOLARPILOT_ENABLE_CUDA=ON requires a CUDA compiler (nvcc). Set CUDACXX or CMAKE_CUDA_COMPILER.")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA target architectures")
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
# The release's top-level CMake builds wxWidgets GUI. Include only coretrace.
add_subdirectory(SolTrace/coretrace EXCLUDE_FROM_ALL)
# SSC's solarpilot-develop superbuild contains only shared/core/nlopt/lpsolve.
set(ENV{CORETRACEDIR} "${CMAKE_CURRENT_SOURCE_DIR}/SolTrace/coretrace")
add_subdirectory(ssc EXCLUDE_FROM_ALL)
if(UNIX)
  # Bundled NLopt expects an Autoconf-generated header on Linux/macOS.
  set(nlopt_config_dir "${CMAKE_BINARY_DIR}/nlopt-config")
  file(MAKE_DIRECTORY "${nlopt_config_dir}")
  if(NOT EXISTS "${nlopt_config_dir}/config.h")
    execute_process(COMMAND "${CMAKE_COMMAND}" -E env
      "CC=${CMAKE_C_COMPILER}" "CXX=${CMAKE_CXX_COMPILER}"
      "${CMAKE_CURRENT_SOURCE_DIR}/ssc/nlopt/configure"
      WORKING_DIRECTORY "${nlopt_config_dir}"
      RESULT_VARIABLE nlopt_config_status
      OUTPUT_FILE "${nlopt_config_dir}/configure.log"
      ERROR_FILE "${nlopt_config_dir}/configure-error.log")
    if(NOT nlopt_config_status EQUAL 0)
      message(FATAL_ERROR "NLopt configure failed; see ${nlopt_config_dir}/configure*.log")
    endif()
  endif()
  target_include_directories(nlopt PRIVATE "${nlopt_config_dir}")
endif()
# CoPylot uses weather parsing and util only, not SSC's PV/battery/MPFIT code.
set_property(TARGET shared PROPERTY SOURCES
  "${CMAKE_CURRENT_SOURCE_DIR}/ssc/shared/lib_weatherfile.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/ssc/shared/lib_util.cpp")
if(BUILD_COPILOT_API)
  if(SOLARPILOT_ENABLE_CUDA)
    target_sources(solarpilot PRIVATE ssc/solarpilot/CudaProbe.cu ssc/solarpilot/FluxDensityKernels.cu)
    target_compile_definitions(solarpilot PRIVATE SOLARPILOT_ENABLE_CUDA)
    target_link_libraries(solarpilot CUDA::cudart)
    set_target_properties(solarpilot PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  endif()
  if(SOLARPILOT_ENABLE_CORE_PROFILE)
    target_sources(solarpilot PRIVATE ssc/solarpilot/CoreProfile.cpp)
    target_compile_definitions(solarpilot PRIVATE SOLARPILOT_ENABLE_CORE_PROFILE)
  endif()
  if(SOLARPILOT_ENABLE_FLUX_DIAGNOSTICS)
    target_compile_definitions(solarpilot PRIVATE SOLARPILOT_ENABLE_FLUX_DIAGNOSTICS)
  endif()
  target_link_libraries(solarpilot Threads::Threads)
  if(UNIX AND NOT APPLE)
    target_link_options(solarpilot PRIVATE -Wl,--no-undefined)
  endif()
  set_target_properties(solarpilot PROPERTIES
    DEBUG_POSTFIX ""
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  include(CTest)
  if(BUILD_TESTING)
    if(SOLARPILOT_ENABLE_CUDA)
      add_executable(cuda_flux_numerics tests/cuda_flux_numerics.cpp)
      target_include_directories(cuda_flux_numerics PRIVATE ssc/solarpilot)
      target_link_libraries(cuda_flux_numerics PRIVATE solarpilot)
      add_test(NAME cuda_flux_numerics COMMAND cuda_flux_numerics)
      set_tests_properties(cuda_flux_numerics PROPERTIES SKIP_RETURN_CODE 77)
    endif()
    add_executable(flux_density_numerics tests/flux_density_numerics.cpp)
    target_include_directories(flux_density_numerics PRIVATE ssc/solarpilot)
    target_link_libraries(flux_density_numerics PRIVATE solarpilot)
    add_test(NAME flux_density_numerics COMMAND flux_density_numerics)
    add_executable(copilot_native_smoke tests/native_smoke.cpp)
    target_link_libraries(copilot_native_smoke PRIVATE solarpilot)
    target_compile_definitions(copilot_native_smoke PRIVATE
      COPILOT_WEATHER="${CMAKE_CURRENT_SOURCE_DIR}/SolarPILOT/deploy/climate_files/USA\ CA\ Daggett\ \(TMY2\).csv")
    add_test(NAME copilot_native_smoke COMMAND copilot_native_smoke)
  endif()
endif()
